function organGroup(THREE: ThreeModule, root: THREE.Group, id: string, color: number) {
  const group = new THREE.Group()
  group.userData.organId = id
  root.add(group)
  const material = new THREE.MeshPhysicalMaterial({ color, roughness: 0.42, metalness: 0.02, clearcoat: 0.22, clearcoatRoughness: 0.42, emissive: 0x000000 })
  const ellipsoid = (x: number, y: number, z: number, sx: number, sy: number, sz: number, materialOverride: THREE.MeshStandardMaterial | THREE.MeshPhysicalMaterial = material) => {
    const mesh = new THREE.Mesh(new THREE.SphereGeometry(1, 40, 28), materialOverride)
    mesh.position.set(x, y, z)
    mesh.scale.set(sx, sy, sz)
    mesh.userData.organId = id
    group.add(mesh)
    return mesh
  }
  return { group, material, ellipsoid }
}

  controls.enablePan = false

    const m = new THREE.Mesh(new THREE.SphereGeometry(1, 20, 14), shellMaterial)
    m.position.set(x, y, z); m.scale.set(sx, sy, sz); shell.add(m)

  silhouette(0, 2.45, 0, .37, .45, .33); silhouette(0, 1.87, 0, .17, .25, .18)

  const bone = new THREE.MeshStandardMaterial({ color: boneColor, roughness: .64, emissive: 0x000000 })
  const bones = organGroup(THREE, root, 'skeleton', boneColor)

    const d = to.clone().sub(from)
    const mesh = new THREE.Mesh(new THREE.CylinderGeometry(radius * .82, radius, d.length(), 12), mat)
    mesh.position.copy(from).add(to).multiplyScalar(.5)
    mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), d.normalize())
    mesh.userData.organId = 'skeleton'; parent.add(mesh)
  }
  line(new THREE.Vector3(0, -.2, -.05), new THREE.Vector3(0, 1.72, -.05), .095)
  // A segmented spine and individually curved ribs replace the old handful of flat hoops.
  for (let i = 0; i < 13; i++) {
    const y = 1.54 - i * .125
    const body = new THREE.Mesh(new THREE.SphereGeometry(1, 20, 14), bone)
    body.position.set(0, y, -.01); body.scale.set(.12 - Math.max(0, i - 9) * .004, .066, .1)
    body.userData.organId = 'skeleton'; bones.group.add(body)
    const process = new THREE.Mesh(new THREE.ConeGeometry(.055, .16, 10), bone)
    process.position.set(0, y, -.17); process.rotation.x = Math.PI / 2; process.userData.organId = 'skeleton'; bones.group.add(process)
  }
  for (let i = 0; i < 10; i++) {
    const y = 1.49 - i * .105
    const width = .3 + Math.sin((i / 10) * Math.PI) * .23
    for (const side of [-1, 1]) {
      const ribCurve = new THREE.CatmullRomCurve3([
        new THREE.Vector3(side * .06, y, -.04),
        new THREE.Vector3(side * width * .68, y + .09, .04),
        new THREE.Vector3(side * width, y + .01, .14),
        new THREE.Vector3(side * width * .84, y - .08, .26),
        new THREE.Vector3(side * .07, y - .12, .29),

      const rib = new THREE.Mesh(new THREE.TubeGeometry(ribCurve, 24, .026, 9, false), bone)
      rib.userData.organId = 'skeleton'; bones.group.add(rib)
    }
  }
  line(new THREE.Vector3(-.48, 1.62, 0), new THREE.Vector3(.48, 1.62, 0), .075)
  line(new THREE.Vector3(-.4, -.15, -.02), new THREE.Vector3(.4, -.15, -.02), .12)
  for (const s of [-1, 1]) {
    line(new THREE.Vector3(s * .5, 1.58, 0), new THREE.Vector3(s * .98, .95, 0), .09)
    line(new THREE.Vector3(s * .98, .95, 0), new THREE.Vector3(s * 1.17, .25, 0), .065)
    line(new THREE.Vector3(s * .28, -.3, 0), new THREE.Vector3(s * .32, -1.05, 0), .13)
    line(new THREE.Vector3(s * .32, -1.05, 0), new THREE.Vector3(s * .34, -1.95, 0), .09)
    // Shoulder, elbow, wrist, knee and ankle landmarks make the long bones read as a connected skeleton.
    for (const [x, y, z, radius] of [[s * .53, 1.56, 0, .14], [s * .99, .91, 0, .105], [s * 1.17, .22, 0, .085], [s * .32, -1.03, 0, .13], [s * .34, -1.92, 0, .1]] as const) {
      const joint = new THREE.Mesh(new THREE.SphereGeometry(radius, 20, 14), bone)
      joint.position.set(x, y, z); joint.userData.organId = 'skeleton'; bones.group.add(joint)
    }
  }
  // Hands, feet, clavicles and sternum give the skeleton a complete, readable silhouette.
  const clavicleCurve = (side: number) => new THREE.CatmullRomCurve3([
    new THREE.Vector3(0, 1.56, .12), new THREE.Vector3(side * .19, 1.64, .13),
    new THREE.Vector3(side * .36, 1.62, .08), new THREE.Vector3(side * .53, 1.55, 0),
  ])
  for (const side of [-1, 1]) {
    const clavicle = new THREE.Mesh(new THREE.TubeGeometry(clavicleCurve(side), 28, .045, 10, false), bone)
    clavicle.userData.organId = 'skeleton'; bones.group.add(clavicle)
    // Radius, ulna and metacarpals; phalanges are short enough to remain legible at full-body scale.
    line(new THREE.Vector3(side * 1.17, .25, 0), new THREE.Vector3(side * 1.24, -.04, .02), .035)
    for (let finger = 0; finger < 5; finger++) {
      const spread = (finger - 2) * .075
      const base = new THREE.Vector3(side * (1.24 + spread), -.04, .02)
      const tip = new THREE.Vector3(side * (1.27 + spread * 1.4), -.27 + Math.abs(finger - 2) * .035, .02)
      line(base, tip, .018)
      if (finger < 4) line(tip, new THREE.Vector3(tip.x + side * .01, tip.y - .095, .02), .014)
    }
    // Tarsals, metatarsals and toes.
    line(new THREE.Vector3(side * .34, -1.95, 0), new THREE.Vector3(side * .35, -2.79, .1), .07)

    },
    zoom(factor: number) {
      const offset = camera.position.clone().sub(controls.target)
      const nextDistance = THREE.MathUtils.clamp(offset.length() * factor, controls.minDistance, controls.maxDistance)
      offset.setLength(nextDistance)
      camera.position.copy(controls.target).add(offset)
      controls.update()
    },
    resetView() { controls.target.set(0, .15, 0); camera.position.set(0, .5, 9.4); controls.update() },
    setSelection(id: string) {
      root.traverse((node) => {
        const object = node as THREE.Mesh
        if (!object.isMesh || !object.material) return
        const materials = Array.isArray(object.material) ? object.material : [object.material]
        materials.forEach((material) => {
          const mat = material as THREE.MeshStandardMaterial
          if (!mat.emissive) return
          mat.emissive.setHex(object.userData.organId === id ? 0x8b743c : 0x000000)
          mat.emissiveIntensity = object.userData.organId === id ? .32 : 0
        })
      })
    },
  }
}
